    % Top Node: The Hidden Port Dilemma

    \node[block, below left=1.2cm and 0.5cm of device, text width=5.2cm] (documented) {\textbf{Documented Interfaces}\\Web Admin UI (Port 80/443)\\Described in user manuals\\Standard credentials / Auth required};

    \node[amberblock, below right=1.2cm and 0.5cm of device, text width=5.2cm] (hidden) {\textbf{Undocumented Services}\\Hidden Telnet / SSH Ports\\Completely omitted from user manuals!\\Hardcoded credentials or vendor backdoors};

    \node[block, below=1.2cm of documented, text width=5.2cm] (testing) {\textbf{Factory Testing \& Debugging}\\Left behind by engineers for QA/maintenance;\\Forgotten before commercial release};

    \draw[->, thick] (hidden) -- (threat);

    % Bottom Synthesis
    \node[greenblock, below=1.0cm of $(testing)!0.5!(threat)$, text width=11.5cm] (remedy) {\textbf{CSS 2025 Ethical Practice Award Finding (YNU SPIRAL)}:\\Distinguishing intentional debug backdoors from accidental security holes;\\Pushing responsible vendor notification, zero-trust firmware inspection, and strict QA standards};
