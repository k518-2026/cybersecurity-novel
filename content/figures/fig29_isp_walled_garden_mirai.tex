\begin{tikzpicture}[

    >=Stealth,
    node distance=1.2cm,
    block/.style={rectangle, draw=blue!70!black, fill=blue!5, rounded corners=4pt, thick, align=center, inner sep=7pt},
    alertblock/.style={rectangle, draw=red!70!black, fill=red!5, rounded corners=4pt, thick, align=center, inner sep=7pt},
    greenblock/.style={rectangle, draw=green!60!black, fill=green!5, rounded corners=4pt, thick, align=center, inner sep=7pt},
    amberblock/.style={rectangle, draw=orange!80!black, fill=orange!5, rounded corners=4pt, thick, align=center, inner sep=7pt},
    groupbox/.style={draw=black!40, dashed, rounded corners=6pt, fill=black!2, inner sep=10pt}
]

    % Top Node: Mirai Infected Consumer IoT
    \node[alertblock] (infected) {\textbf{Infected Consumer IoT Devices}\\Routers, Cameras, DVRs in Home Networks};

    % Central Comparison: Two Intervention Methods
    \node[amberblock, below left=1.2cm and 0.5cm of infected, text width=5.2cm] (email) {\textbf{Email Notification Alone}\\Ignored as spam, technical barrier\\Result: \textbf{Zero statistically significant effect}};

    \node[greenblock, below right=1.2cm and 0.5cm of infected, text width=5.2cm] (walled) {\textbf{ISP Walled Garden Isolation}\\Captive browser redirection on web access\\Result: \textbf{92\% remediated within 14 days!}};

    \draw[->, thick, red!70!black] (infected) -| node[above, font=\footnotesize] {Passive Notification} (email);
    \draw[->, thick, green!60!black] (infected) -| node[above, font=\footnotesize] {Network Quarantine} (walled);

    % Mental Model vs Action
    \node[block, below=1.2cm of email, text width=5.2cm] (mental) {\textbf{Flawed Consumer Mental Models}\\Ran PC antivirus, rebooted router;\\Often lacked understanding of IoT root causes};

    \node[greenblock, below=1.2cm of walled, text width=5.2cm] (enforcement) {\textbf{Guided Automatic Remediation}\\ISP captive portal guides password reset\\Forces prompt action without disruption};

    \draw[->, thick] (email) -- (mental);
    \draw[->, thick] (walled) -- (enforcement);

    % Bottom Synthesis
    \node[block, fill=blue!10, below=1.0cm of $(mental)!0.5!(enforcement)$, text width=11.5cm] (summary) {\textbf{Key Insight (NDSS 2019 Distinguished Paper)}:\\Consumer awareness campaigns fail alone; ISP-level network intervention (Walled Garden) is essential for eradicating global IoT botnets};
    \draw[->, thick] (mental) |- (summary);
    \draw[->, thick] (enforcement) |- (summary);

    \begin{scope}[on background layer]
        \node[groupbox, fit=(infected) (email) (walled) (mental) (enforcement) (summary), label={[anchor=north west, font=\footnotesize\bfseries, blue!80!black]north west:NDSS 2019: Real-World Evidence on ISP vs Consumer Efforts}] {};
    \end{scope}

\end{tikzpicture}
